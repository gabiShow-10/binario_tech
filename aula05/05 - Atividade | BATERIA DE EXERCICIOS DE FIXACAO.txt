Exercicio 04:
Crie um middleware exclusivo de validacao de CNH chamado 'middlewares/validaCnh.js' que garante que a CNH informada no POST contenha exatamente 11 digitos numericos.
function validaCnh(req, res, next) {
    const { cnh } = req.body;
    if (!cnh || !/^\d{11}$/.test(cnh)) {
        return res.status(400).json({ erro: "A CNH deve conter exatamente 11 dígitos numéricos." });
    }
    next();
}
module.exports = validaCnh;

adicionado ao arquivo middlewares/validaCnh.js
